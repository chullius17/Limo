cmake_minimum_required(VERSION 3.8)
project(limo_smac_planner)
find_package(ament_cmake REQUIRED)
find_package(nav2_core REQUIRED)
find_package(nav2_costmap_2d REQUIRED)
find_package(nav2_util REQUIRED)
find_package(rclcpp REQUIRED)
find_package(pluginlib REQUIRED)
find_package(tf2_geometry_msgs REQUIRED)
find_package(limo_interfaces REQUIRED)
find_package(ompl REQUIRED)
add_library(${PROJECT_NAME} SHARED src/explicit_start_planner.cpp src/dubins_connector.cpp)
target_compile_features(${PROJECT_NAME} PUBLIC cxx_std_17)
target_include_directories(${PROJECT_NAME} PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:include> ${OMPL_INCLUDE_DIRS})
ament_target_dependencies(${PROJECT_NAME}
  nav2_core nav2_costmap_2d nav2_util rclcpp pluginlib tf2_geometry_msgs limo_interfaces)
target_link_libraries(${PROJECT_NAME} ${OMPL_LIBRARIES})
pluginlib_export_plugin_description_file(nav2_core plugins.xml)
install(TARGETS ${PROJECT_NAME} LIBRARY DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
install(FILES plugins.xml DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_dubins_connector test/test_dubins_connector.cpp)
  target_link_libraries(test_dubins_connector ${PROJECT_NAME})
  ament_add_gtest(test_explicit_start_planner test/test_explicit_start_planner.cpp)
  target_link_libraries(test_explicit_start_planner ${PROJECT_NAME})
endif()
ament_export_include_directories(include)
ament_export_libraries(${PROJECT_NAME})
ament_export_dependencies(nav2_core nav2_costmap_2d nav2_util rclcpp pluginlib
  tf2_geometry_msgs limo_interfaces ompl)
ament_package()
